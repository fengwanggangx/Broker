cmake_minimum_required(VERSION 3.20)
project(Broker LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
enable_testing()
add_subdirectory(counter)
option(BROKER_BUILD_SERVICE "Build standalone Broker service" ON)
if(BROKER_BUILD_SERVICE)
 set(BROKER_DEPS "${CMAKE_CURRENT_SOURCE_DIR}" CACHE PATH "Broker dependency bundle")
 set(BROKER_MODE "debug" CACHE STRING "Dependency configuration: debug or release")
 set(depInc "${BROKER_DEPS}/include/${BROKER_MODE}")
 set(depLib "${BROKER_DEPS}/libs/${BROKER_MODE}")
 file(GLOB supportSources CONFIGURE_DEPENDS basic/*.cpp common/*.cpp database/*.cpp ini/*.cpp network/*.cpp thread/*.cpp)
 file(GLOB protobufSources CONFIGURE_DEPENDS request/*.cc request/v1/*.cc)
 file(GLOB abslLibraries "${depLib}/libabsl_*.so")
 add_library(broker_support STATIC ${supportSources} ${protobufSources} request/request.cpp request/RequestCenter.cpp request/MarketTypes.cpp system/CSession.cpp system/CHostMgr.cpp)
 target_include_directories(broker_support PUBLIC "${depInc}/protobuf" "${depInc}/libevent" "${depInc}/mysql" "${depInc}/mysql/mysql" "${depInc}/sqlite" "${depInc}/rapidjson" "${depInc}/ini" request)
 target_link_directories(broker_support PUBLIC "${depLib}")
 set(protoSuffix "")
 if(BROKER_MODE STREQUAL "debug")
  set(protoSuffix "d")
 endif()
 target_link_libraries(broker_support PUBLIC protobuf${protoSuffix} ${abslLibraries} utf8_validity utf8_range mysqlclient sqlite3 event event_core event_extra event_openssl event_pthreads Threads::Threads)
 add_executable(Broker main.cpp system/CBootLoader.cpp service/CTradeService.cpp system/CBrokerService.cpp counter/CounterSetup.cpp counter/CStorager.cpp counter/TradeProtocol.cpp)
 target_link_libraries(Broker PRIVATE trading_counter broker_support)
 set_target_properties(Broker PROPERTIES BUILD_RPATH "${depLib}" INSTALL_RPATH "$ORIGIN/../../libs/${BROKER_MODE}")
endif()

if(BROKER_BUILD_SERVICE)
 add_executable(counter_protocol_tests counter/tests/TradeProtocolTests.cpp counter/TradeProtocol.cpp)
 target_link_libraries(counter_protocol_tests PRIVATE trading_counter broker_support)
 add_test(NAME counter_protocol_tests COMMAND counter_protocol_tests)
endif()

set(CMAKE_BUILD_RPATH "${depLib}")
