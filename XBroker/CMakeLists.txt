cmake_minimum_required(VERSION 3.20)
project(XBroker LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
enable_testing()
add_subdirectory(counter)
option(XBROKER_BUILD_SERVICE "Build standalone XBroker service" ON)
if(XBROKER_BUILD_SERVICE)
 set(XBROKER_DEPS "${CMAKE_CURRENT_SOURCE_DIR}" CACHE PATH "XBroker dependency bundle")
 set(XBROKER_MODE "debug" CACHE STRING "Dependency configuration: debug or release")
 set(depInc "${XBROKER_DEPS}/include/${XBROKER_MODE}")
 set(depLib "${XBROKER_DEPS}/libs/${XBROKER_MODE}")
 file(GLOB supportSources CONFIGURE_DEPENDS basic/*.cpp common/*.cpp database/*.cpp ini/*.cpp network/*.cpp thread/*.cpp)
 file(GLOB protobufSources CONFIGURE_DEPENDS request/*.cc request/v1/*.cc)
 file(GLOB abslLibraries "${depLib}/libabsl_*.so")
 add_library(xbroker_support STATIC ${supportSources} ${protobufSources} request/request.cpp request/RequestCenter.cpp request/MarketTypes.cpp system/CSession.cpp system/CHostMgr.cpp)
 target_include_directories(xbroker_support PUBLIC "${depInc}/protobuf" "${depInc}/libevent" "${depInc}/mysql" "${depInc}/mysql/mysql" "${depInc}/sqlite" "${depInc}/rapidjson" "${depInc}/ini" request)
 target_link_directories(xbroker_support PUBLIC "${depLib}")
 set(protoSuffix "")
 if(XBROKER_MODE STREQUAL "debug")
  set(protoSuffix "d")
 endif()
 target_link_libraries(xbroker_support PUBLIC protobuf${protoSuffix} ${abslLibraries} utf8_validity utf8_range mysqlclient sqlite3 event event_core event_extra event_openssl event_pthreads Threads::Threads)
 add_executable(XBroker main.cpp system/CBootLoader.cpp service/CTradeService.cpp system/CBrokerService.cpp counter/CounterSetup.cpp counter/CSimulatedCounterStorage.cpp counter/CSimulatedCounterMarket.cpp counter/CStorager.cpp counter/TradeProtocol.cpp)
 target_link_libraries(XBroker PRIVATE trading_counter xbroker_support)
 set_target_properties(XBroker PROPERTIES BUILD_RPATH "${depLib}" INSTALL_RPATH "$ORIGIN/../../libs/${XBROKER_MODE}")
endif()

if(XBROKER_BUILD_SERVICE)
 add_executable(counter_protocol_tests counter/tests/TradeProtocolTests.cpp counter/TradeProtocol.cpp)
 target_link_libraries(counter_protocol_tests PRIVATE trading_counter xbroker_support)
 add_test(NAME counter_protocol_tests COMMAND counter_protocol_tests)
endif()

set(CMAKE_BUILD_RPATH "${depLib}")

if(XBROKER_BUILD_SERVICE)
 add_executable(login_service_tests counter/tests/LoginServiceTests.cpp service/CTradeService.cpp counter/CounterSetup.cpp counter/CSimulatedCounterStorage.cpp counter/CSimulatedCounterMarket.cpp counter/CStorager.cpp counter/TradeProtocol.cpp)
 target_link_libraries(login_service_tests PRIVATE trading_counter xbroker_support)
 add_test(NAME login_service_tests COMMAND login_service_tests)
endif()

option(XBROKER_ENABLE_XTP "Enable ZhongTai traditional XTP 2.2.50.8" OFF)
set(XBROKER_XTP_SDK "" CACHE PATH "External SDK root containing include and linux_glibc_2.17")
if(XBROKER_BUILD_SERVICE)
 add_library(zhongtai_adapter STATIC counter/zhongtai/CZhongTaiGateway.cpp counter/zhongtai/CZhongTaiCounter.cpp counter/zhongtai/CZhongTaiApi.cpp counter/zhongtai/CZhongTaiStore.cpp counter/zhongtai/ZhongTaiConvert.cpp counter/zhongtai/ZhongTaiSetup.cpp)
 target_link_libraries(zhongtai_adapter PUBLIC trading_counter xbroker_support)
 target_link_libraries(XBroker PRIVATE zhongtai_adapter)
 target_link_libraries(login_service_tests PRIVATE zhongtai_adapter)
 if(XBROKER_ENABLE_XTP)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$")
   message(FATAL_ERROR "ZhongTai SDK requires Linux x64")
  endif()
  set(xtpHeader "${XBROKER_XTP_SDK}/include/xtp_trader_api.h")
  set(xtpLibrary "${XBROKER_XTP_SDK}/linux_glibc_2.17/libxtptraderapi.so")
  if(NOT EXISTS "${xtpHeader}" OR NOT EXISTS "${xtpLibrary}")
   message(FATAL_ERROR "XTP enabled but SDK missing; set XBROKER_XTP_SDK")
  endif()
  file(SHA256 "${xtpHeader}" xtpHeaderHash)
  file(SHA256 "${xtpLibrary}" xtpLibraryHash)
  if(NOT xtpHeaderHash STREQUAL "c55b053f21be35c03b734e7ee9774d86b771b27d23d5fa6d562869aa85eb7874" OR NOT xtpLibraryHash STREQUAL "f6d4b688c10306e1699330892c4275c8f8cb36fda541c8ced28fd23713f364ee")
   message(FATAL_ERROR "XTP SDK checksum does not match official 2.2.50.8")
  endif()
  target_compile_definitions(zhongtai_adapter PRIVATE XBROKER_ENABLE_XTP)
  target_include_directories(zhongtai_adapter SYSTEM PRIVATE "${XBROKER_XTP_SDK}/include")
  target_link_libraries(zhongtai_adapter PUBLIC "${xtpLibrary}")
 endif()
endif()

if(XBROKER_BUILD_SERVICE)
 add_executable(zhongtai_tests counter/tests/ZhongTaiTests.cpp counter/TradeProtocol.cpp)
 target_link_libraries(zhongtai_tests PRIVATE zhongtai_adapter)
 add_test(NAME zhongtai_tests COMMAND zhongtai_tests)
 set_tests_properties(zhongtai_tests PROPERTIES TIMEOUT 30)
endif()

 add_executable(zhongtai_store_tests counter/tests/ZhongTaiStoreTests.cpp)

 add_test(NAME zhongtai_store_tests COMMAND zhongtai_store_tests)
 set_tests_properties(zhongtai_store_tests PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 30)
endif()
